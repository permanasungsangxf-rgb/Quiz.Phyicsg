        function togglePasscodeVisibility() {
            const passInput = document.getElementById('access-code');
            const passIcon = document.getElementById('pass-icon');
            if (passInput.type === 'password') {
                passInput.type = 'text';
                passIcon.classList.remove('fa-eye');
                passIcon.classList.add('fa-eye-slash');
            } else {
                passInput.type = 'password';
                passIcon.classList.remove('fa-eye-slash');
                passIcon.classList.add('fa-eye');
            }
        }

        function startQuiz() {
            const nameInput = document.getElementById('player-name');
            const codeInput = document.getElementById('access-code');
            const nameError = document.getElementById('name-error');
            const errorMsgText = document.getElementById('error-msg-text');
            const intruderAlert = document.getElementById('intruder-alert');
            const intruderTitle = document.getElementById('intruder-title');
            const intruderDesc = document.getElementById('intruder-desc');

            // 1. Cek Nama kosong
            if (!nameInput.value.trim()) {
                errorMsgText.textContent = "Isi nama atau nickname kamu dulu ya!";
                nameError.classList.remove('hidden');
                intruderAlert.classList.add('hidden');
                return;
            }

            const inputCode = codeInput.value.trim().toUpperCase();

            // 2. Cek apakah kode sudah pernah dipakai (Single-Use Validation)
            let usedCodes = JSON.parse(localStorage.getItem('physics_quiz_used_codes') || '[]');
            if (usedCodes.includes(inputCode)) {
                errorMsgText.textContent = "Kode akses ini sudah pernah digunakan!";
                nameError.classList.remove('hidden');
                intruderTitle.textContent = "KODE AKSES SUDAH HANGUS / EXPIRED!";
                intruderDesc.textContent = "Kode akses ini bersifat sekali pakai dan sudah digunakan sebelumnya. Minta kode baru ke Guru!";
                intruderAlert.classList.remove('hidden');
                return;
            }

            // 3. Cek Kebenaran Kode Akses
            if (inputCode !== MASTER_ACCESS_CODE) {
                errorMsgText.textContent = "Kode akses salah/tidak valid!";
                nameError.classList.remove('hidden');
                intruderTitle.textContent = "PERINGATAN PENYUSUP DITETAPKAN!";
                intruderDesc.textContent = "Kode akses salah. Akses ditolak & percobaan dicatat ke log proteksi!";
                intruderAlert.classList.remove('hidden');

                // Simpan rekam jejak penyusup ke localStorage log
                let intruderLogs = JSON.parse(localStorage.getItem('physics_quiz_intruders') || '[]');
                intruderLogs.push({
                    name: nameInput.value.trim(),
                    attemptCode: codeInput.value.trim(),
                    timestamp: new Date().toLocaleString('id-ID')
                });
                localStorage.setItem('physics_quiz_intruders', JSON.stringify(intruderLogs));
                return;
            }

            // Tandai kode ini sebagai SUDAH DIPAKAI (Hangus / Single-Use)
            usedCodes.push(inputCode);
            localStorage.setItem('physics_quiz_used_codes', JSON.stringify(usedCodes));

            nameError.classList.add('hidden');
            intruderAlert.classList.add('hidden');
            playerName = nameInput.value.trim();

            // Reset state
            currentQuestionIndex = 0;
            score = 0;
            totalSeconds = 0;
            userAnswersHistory = [];

            document.getElementById('live-score').textContent = score;
            document.getElementById('total-q').textContent = quizData.length;

            // Switch screens
            document.getElementById('welcome-screen').classList.add('hidden');
            document.getElementById('quiz-screen').classList.remove('hidden');

            // Start timer
            startTimer();

            // Load first question
            loadQuestion();
        }

        function startTimer() {
            clearInterval(timerInterval);
            totalSeconds = 0;
            updateTimerDisplay();

            timerInterval = setInterval(() => {
                totalSeconds++;
                updateTimerDisplay();
            }, 1000);
        }

        function updateTimerDisplay() {
            let mins = Math.floor(totalSeconds / 60).toString().padStart(2, '0');
            let secs = (totalSeconds % 60).toString().padStart(2, '0');
            document.getElementById('timer-display').textContent = `${mins}:${secs}`;
        }

            document.getElementById('question-difficulty').textContent = q.difficulty;

                btn.className = `w-full text-left p-4 rounded-2xl bg-slate-900/80 hover:bg-indigo-950/80 border border-indigo-500/20 hover:border-cyan-400/60 text-slate-200 text-sm sm:text-base font-medium transition flex items-center justify-between group shadow-sm`;
                btn.innerHTML = `
                    <div class="flex items-center space-x-3">
                        <span class="w-7 h-7 rounded-xl bg-slate-800 border border-indigo-500/30 flex items-center justify-center text-xs font-bold text-cyan-300 group-hover:bg-cyan-500 group-hover:text-slate-950 transition flex-shrink-0">
                            ${String.fromCharCode(65 + idx)}
                        </span>
                        <span class="leading-relaxed">${opt}</span>
                    </div>
                    <i class="fa-solid fa-chevron-right text-slate-600 group-hover:text-cyan-400 transition text-xs"></i>
                `;
                btn.onclick = () => selectAnswer(idx, btn);
                optionsContainer.appendChild(btn);
            });
        }

        function selectAnswer(selectedIndex, selectedBtn) {
            const q = quizData[currentQuestionIndex];
            const optionsContainer = document.getElementById('options-container');
            const buttons = optionsContainer.getElementsByTagName('button');

            // Nonaktifkan semua tombol setelah dipilih
            for (let b of buttons) {
                b.disabled = true;
                b.classList.remove('hover:bg-slate-800/90', 'hover:border-indigo-500/50');
            }

            const isCorrect = selectedIndex === q.correct;
            if (isCorrect) {
                score += 5; // 20 soal x 5 poin = Max 100 poin
                document.getElementById('live-score').textContent = score;
                selectedBtn.classList.remove('bg-slate-900/70', 'border-slate-800');
                selectedBtn.classList.add('bg-emerald-500/20', 'border-emerald-500/60', 'text-emerald-200');
                selectedBtn.querySelector('span').classList.add('bg-emerald-600', 'text-white', 'border-emerald-500');
            } else {
                selectedBtn.classList.remove('bg-slate-900/70', 'border-slate-800');
                selectedBtn.classList.add('bg-rose-500/20', 'border-rose-500/60', 'text-rose-200');
                selectedBtn.querySelector('span').classList.add('bg-rose-600', 'text-white', 'border-rose-500');

                // Tunjukkan yang benar
                buttons[q.correct].classList.remove('bg-slate-900/70', 'border-slate-800');
                buttons[q.correct].classList.add('bg-emerald-500/20', 'border-emerald-500/60', 'text-emerald-200');
                buttons[q.correct].querySelector('span').classList.add('bg-emerald-600', 'text-white', 'border-emerald-500');
            }

            userAnswersHistory.push({ question: q.question, correct: isCorrect });

            // Tampilkan kotak penjelasan
            document.getElementById('explanation-text').textContent = q.explanation;
            document.getElementById('explanation-card').classList.remove('hidden');
        }

        function nextQuestion() {
            currentQuestionIndex++;
            if (currentQuestionIndex < quizData.length) {
                loadQuestion();
            } else {
                endQuiz();
            }
        }

        function endQuiz() {
            clearInterval(timerInterval);
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('result-screen').classList.remove('hidden');

            document.getElementById('result-name').textContent = playerName;
            document.getElementById('final-score').textContent = score;
            
            const correctCount = userAnswersHistory.filter(a => a.correct).length;
            const accuracy = Math.round((correctCount / quizData.length) * 100);
            document.getElementById('final-accuracy').textContent = `${accuracy}%`;

            let mins = Math.floor(totalSeconds / 60).toString().padStart(2, '0');
            let secs = (totalSeconds % 60).toString().padStart(2, '0');
            const timeStr = `${mins}:${secs}`;
            document.getElementById('final-time').textContent = timeStr;

            // Simpan ke Leaderboard
            saveToLeaderboard(playerName, score, correctCount, timeStr);
        }

        function saveToLeaderboard(name, scorePoin, correctCount, time) {
            let lb = JSON.parse(localStorage.getItem('physics_quiz_leaderboard') || '[]');
            lb.push({ name: name, score: scorePoin, correct: correctCount, time: time });
            // Urutkan berdasarkan skor tertinggi, lalu waktu tercepat
            lb.sort((a, b) => b.score - a.score || a.time.localeCompare(b.time));
            // Ambil top 10
            lb = lb.slice(0, 10);
            localStorage.setItem('physics_quiz_leaderboard', JSON.stringify(lb));
        }

        function restartQuiz() {
            document.getElementById('result-screen').classList.add('hidden');
            document.getElementById('welcome-screen').classList.remove('hidden');
            document.getElementById('player-name').value = '';
        }

            const listContainer = document.getElementById('leaderboard-list');

        // System State
        let currentQuestion = 0;
        let score = 0;
        let seconds = 0;
        let timer = null;
        let userName = "";

        // Memory Key for Used Passcodes
        const USED_CODES_KEY = "physics_quiz_used_codes";

        function getUsedCodes() {
            const data = localStorage.getItem(USED_CODES_KEY);
            return data ? JSON.parse(data) : [];
        }

        function saveUsedCode(code) {
            const used = getUsedCodes();
            used.push(code.toUpperCase());
            localStorage.setItem(USED_CODES_KEY, JSON.stringify(used));
        }

        // Generate New One-Time Code for Teachers
        function generateNewPasscode() {
            const randomCode = "FISIKA-" + Math.random().toString(36).substring(2, 6).toUpperCase();
            document.getElementById('generated-code').innerText = randomCode;
            document.getElementById('new-code-display').classList.remove('hidden');
        }

        // Start Validation
        function validateAndStart() {
            const nameInput = document.getElementById('input-name').value.trim();
            const codeInput = document.getElementById('input-code').value.trim().toUpperCase();
            const alertBox = document.getElementById('access-alert');
            const alertMsg = document.getElementById('access-alert-msg');

            if (!nameInput) {
                alertMsg.innerText = "Silakan masukkan nama Anda terlebih dahulu!";
                alertBox.classList.remove('hidden');
                return;
            }

            if (!codeInput) {
                alertMsg.innerText = "Masukkan Kode Akses Kuis!";
                alertBox.classList.remove('hidden');
                return;
            }

            const usedCodes = getUsedCodes();

            if (usedCodes.includes(codeInput)) {
                alertMsg.innerText = "KODE SUDAH KADALUARSA / PERNAH DIPAKAI! Klik 'Buat Kode Baru' di bawah.";
                alertBox.classList.remove('hidden');
                return;
            }

            // Valid passkey check (Default or generated)
            if (codeInput === "FISIKA2026" || codeInput.startsWith("FISIKA-")) {
                saveUsedCode(codeInput); // Mark as used
                userName = nameInput;
                alertBox.classList.add('hidden');
                startQuiz();
            } else {
                alertMsg.innerText = "KODE AKSES SALAH / TIDAK TERDAFTAR!";
                alertBox.classList.remove('hidden');
            }
        }

        function startQuiz() {
            document.getElementById('start-screen').classList.add('hidden');
            document.getElementById('quiz-screen').classList.remove('hidden');
            document.getElementById('quiz-status').classList.remove('hidden');

            timer = setInterval(() => {
                seconds++;
                const mins = Math.floor(seconds / 60).toString().padStart(2, '0');
                const secs = (seconds % 60).toString().padStart(2, '0');
                document.getElementById('timer-display').innerText = `${mins}:${secs}`;
            }, 1000);

            loadQuestion();
        }

        function loadQuestion() {
            const q = quizData[currentQuestion];
            document.getElementById('question-number').innerText = `Soal ${currentQuestion + 1} dari ${quizData.length}`;
            document.getElementById('question-pioneer').innerText = `Tokoh: ${q.pioneer}`;
            document.getElementById('visual-container').innerHTML = q.svg;
            document.getElementById('question-text').innerText = q.text;

            const optsContainer = document.getElementById('options-container');
            optsContainer.innerHTML = '';

            q.options.forEach((opt, idx) => {
                const btn = document.createElement('button');
                btn.className = 'w-full text-left p-4 rounded-2xl bg-slate-900/60 hover:bg-indigo-900/40 border border-slate-800 hover:border-indigo-500/50 text-slate-200 text-sm md:text-base transition flex items-center gap-3';
                btn.innerHTML = `<span class="w-7 h-7 rounded-lg bg-indigo-950 border border-indigo-500/30 flex items-center justify-center font-bold text-xs text-indigo-400">${String.fromCharCode(65 + idx)}</span> <span>${opt}</span>`;
                btn.onclick = () => selectOption(idx);
                optsContainer.appendChild(btn);
            });
        }

        function selectOption(index) {
            if (index === quizData[currentQuestion].correct) {
                score += 5; // 20 questions * 5 = 100 points
            }

            currentQuestion++;

            if (currentQuestion < quizData.length) {
                loadQuestion();
            } else {
                endQuiz();
            }
        }

        function endQuiz() {
            clearInterval(timer);
            document.getElementById('quiz-screen').classList.add('hidden');
            document.getElementById('quiz-status').classList.add('hidden');
            document.getElementById('result-screen').classList.remove('hidden');

            document.getElementById('result-user-info').innerText = `Kerja bagus, ${userName}! Anda telah menyelesaikan kuis pemahaman fisika.`;
            document.getElementById('final-score').innerText = score;
            document.getElementById('final-time').innerText = `${seconds}s`;
        }
